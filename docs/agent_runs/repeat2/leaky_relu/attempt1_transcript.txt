```python
import torch
import triton
import triton.language as tl


@triton.jit
def leaky_relu_bf16_kernel(
    x_ptr,
    out_ptr,
    n_elements,
    negative_slope,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements

    x = tl.load(x_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    y = tl.where(x > 0.0, x, x * negative_slope)
    tl.store(out_ptr + offsets, y.to(tl.bfloat16), mask=mask)


TRITON_KERNELS = [leaky_relu_bf16_kernel]


def run(*tensors: torch.Tensor) -> torch.Tensor:
    a = tensors[0].contiguous()
    out = torch.empty(a.shape, dtype=a.dtype, device=a.device)
    n_elements = a.numel()

    BLOCK_SIZE = 1024
    grid = (triton.cdiv(n_elements, BLOCK_SIZE),)

    leaky_relu_bf16_kernel[grid](
        a,
        out,
        n_elements,
        0.01,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out
```
