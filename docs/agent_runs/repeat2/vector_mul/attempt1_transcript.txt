```python
import torch
import triton
import triton.language as tl


@triton.jit
def vector_mul_kernel(a_ptr, b_ptr, c_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    a = tl.load(a_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    c = a * b
    tl.store(c_ptr + offsets, c.to(tl.bfloat16), mask=mask)


TRITON_KERNELS = [vector_mul_kernel]


def run(*tensors: torch.Tensor) -> torch.Tensor:
    a, b = tensors
    n_elements = a.numel()
    c = torch.empty(a.shape, dtype=a.dtype, device=a.device)
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(n_elements, BLOCK_SIZE),)
    vector_mul_kernel[grid](a, b, c, n_elements, BLOCK_SIZE=BLOCK_SIZE)
    return c
```
