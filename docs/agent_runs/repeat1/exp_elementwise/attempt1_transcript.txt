```python
import torch
import triton
import triton.language as tl


@triton.jit
def exp_elementwise_kernel(a_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offsets < n_elements
    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    x = a.to(tl.float32)
    y = tl.exp(x)
    tl.store(out_ptr + offsets, y.to(out_ptr.dtype.element_ty), mask=mask)


TRITON_KERNELS = [exp_elementwise_kernel]


def run(*tensors: torch.Tensor) -> torch.Tensor:
    a = tensors[0].contiguous()
    out = torch.empty(a.shape, dtype=a.dtype, device=a.device)
    n = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    exp_elementwise_kernel[grid](a, out, n, BLOCK=BLOCK)
    return out
```
