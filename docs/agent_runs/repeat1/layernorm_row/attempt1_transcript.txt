```python
import torch
import triton
import triton.language as tl


@triton.jit
def _layernorm_row_kernel(
    x_ptr,
    out_ptr,
    N,
    stride_xm,
    stride_om,
    eps,
    BLOCK_N: tl.constexpr,
):
    row = tl.program_id(0)
    cols = tl.arange(0, BLOCK_N)
    mask = cols < N

    x = tl.load(x_ptr + row * stride_xm + cols, mask=mask, other=0.0).to(tl.float32)

    inv_n = 1.0 / N
    mean = tl.sum(x, axis=0) * inv_n
    xc = tl.where(mask, x - mean, 0.0)
    var = tl.sum(xc * xc, axis=0) * inv_n
    rstd = 1.0 / tl.sqrt(var + eps)
    y = xc * rstd

    tl.store(
        out_ptr + row * stride_om + cols,
        y.to(out_ptr.dtype.element_ty),
        mask=mask,
    )


TRITON_KERNELS = [_layernorm_row_kernel]


def run(*tensors: torch.Tensor) -> torch.Tensor:
    x = tensors[0].contiguous()
    assert x.dim() == 2, "layernorm_row expects a 2D tensor"
    M, N = x.shape
    out = torch.empty((M, N), dtype=x.dtype, device=x.device)

    BLOCK_N = triton.next_power_of_2(N)
    num_warps = 4 if BLOCK_N <= 1024 else 8

    _layernorm_row_kernel[(M,)](
        x,
        out,
        N,
        x.stride(0),
        out.stride(0),
        1e-5,
        BLOCK_N=BLOCK_N,
        num_warps=num_warps,
    )
    return out
```
