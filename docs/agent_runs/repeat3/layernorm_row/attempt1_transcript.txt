```python
import torch
import triton
import triton.language as tl


@triton.jit
def _layernorm_row_kernel(x_ptr, y_ptr, stride_x, stride_y, N, eps, BLOCK_N: tl.constexpr):
    row = tl.program_id(0)
    offs = tl.arange(0, BLOCK_N)
    mask = offs < N
    x = tl.load(x_ptr + row * stride_x + offs, mask=mask, other=0.0).to(tl.float32)
    mean = tl.sum(x, axis=0) / N
    xc = tl.where(mask, x - mean, 0.0)
    var = tl.sum(xc * xc, axis=0) / N
    rstd = 1.0 / tl.sqrt(var + eps)
    y = xc * rstd
    tl.store(y_ptr + row * stride_y + offs, y.to(y_ptr.dtype.element_ty), mask=mask)


TRITON_KERNELS = [_layernorm_row_kernel]


def run(*tensors: torch.Tensor) -> torch.Tensor:
    x = tensors[0].contiguous()
    M, N = x.shape
    y = torch.empty_like(x)
    BLOCK_N = triton.next_power_of_2(N)
    num_warps = 8 if BLOCK_N >= 2048 else 4
    _layernorm_row_kernel[(M,)](
        x, y, x.stride(0), y.stride(0), N, 1e-5,
        BLOCK_N=BLOCK_N, num_warps=num_warps,
    )
    return y
```
