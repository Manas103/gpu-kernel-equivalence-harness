```python
import torch
import triton
import triton.language as tl


@triton.jit
def _softplus_kernel(x_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    x = tl.load(x_ptr + offsets, mask=mask, other=0.0).to(tl.float32)
    # Numerically stable softplus: max(x, 0) + log(1 + exp(-|x|))
    z = tl.exp(-tl.abs(x))
    y = tl.maximum(x, 0.0) + tl.log(1.0 + z)
    tl.store(out_ptr + offsets, y.to(out_ptr.dtype.element_ty), mask=mask)


TRITON_KERNELS = [_softplus_kernel]


def run(*tensors: torch.Tensor) -> torch.Tensor:
    a = tensors[0]
    x = a.contiguous().view(-1)
    out = torch.empty_like(x)
    n = x.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(n, BLOCK_SIZE),)
    _softplus_kernel[grid](x, out, n, BLOCK_SIZE=BLOCK_SIZE)
    return out.view(a.shape)
```
